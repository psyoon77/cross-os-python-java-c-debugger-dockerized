0 create a GitHub account

1 open powershell as administrator - right click

2 cd C:\Users\[user name]\.ssh

3 generate a new SSH key
ssh-keygen -t ed25519 -C "[comment - recommended key name]" -f ~/.ssh/id_ed25519_[name_of_key]
	add passphrase for the key when prompted

4 notepad config - add the following -
Host github-[name of the key]
  HostName github.com
  User git
  IdentityFile ~/.ssh/id_ed25519_[name of the key]
  IdentitiesOnly yes

5 start ssh agent
Get-Service ssh-agent
Set-Service -Name ssh-agent -StartupType Automatic
Start-Service ssh-agent

6 add key to ssh agent
ssh-add $env:USERPROFILE\.ssh\id_ed25519_[name of the key]

7 copy the public ssh key
Get-Content $env:USERPROFILE\.ssh\id_ed25519_[name of the key].pub

8 upload the public ssh key on the github account
settings - ssh and gpg keys - ssh keys - new ssh key -
	name it [name of the key recommended] - paste the public key

9 test
ssh-add -l
ssh -T github-[name of the key]

10 github cli authentication
gh auth status
gh auth logout
gh auth login
choose github.com
choose ssh for git operations
choose the correct public ssh key that is on the github account
press enter without any input - title for your ssh key:
choose login with a web browser
copy the one-time code
press enter
follow the gui github authentication process
gh auth status
